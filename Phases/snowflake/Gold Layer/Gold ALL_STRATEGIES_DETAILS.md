# Create table 

```


USE DATABASE SALES_ANALYTICS_DB;
USE SCHEMA GOLD;

CREATE OR REPLACE VIEW GOLD.ALL_STRATEGIES_DETAILS AS
SELECT
    LEAD_ID,

    ACTIVITY_AT::DATE AS ACTIVITY_LOG_DATE,

    CUSTOM_ACTIVITY,

    -- CRM record status, such as published or draft
    STATUS_CHANGE AS STATUS,

    CLOSER_EMAIL,
    CLOSER_NAME,

    -- Populated for Strategy Calls;
    -- naturally NULL for Strategy Call Follow Ups
    CUSTOM_ACTIVITY_OUTCOME AS STRATEGY_CALL_OUTCOME,

    ACTIVITY_AT::DATE AS DATE_OF_STRATEGY_CALL,

    OFFER_PRESENTED,

    -- SME model uses SETTER plus SETTER_NAME.
    -- SETTER is represented by the setter's email.
    SETTER_EMAIL AS SETTER,
    SETTER_NAME

FROM SALES_ANALYTICS_DB.SILVER.LEADS_ACTIVITIES_SUMMARY
WHERE CUSTOM_ACTIVITY IN (
    '5) Strategy Call',
    '6) Strategy Call Follow Up'
);

```
# Validate row count

```
SELECT COUNT(*) AS ALL_STRATEGIES_DETAILS_COUNT
FROM GOLD.ALL_STRATEGIES_DETAILS;


```

# Validate source-to-Gold reconciliation

```
SELECT
    CUSTOM_ACTIVITY,
    COUNT(*) AS GOLD_ROWS
FROM GOLD.ALL_STRATEGIES_DETAILS
GROUP BY CUSTOM_ACTIVITY
ORDER BY CUSTOM_ACTIVITY;

```


# Validate outcome distribution

```
SELECT
    CUSTOM_ACTIVITY,
    STRATEGY_CALL_OUTCOME,
    COUNT(*) AS TOTAL
FROM GOLD.ALL_STRATEGIES_DETAILS
GROUP BY
    CUSTOM_ACTIVITY,
    STRATEGY_CALL_OUTCOME
ORDER BY
    CUSTOM_ACTIVITY,
    TOTAL DESC;
```


# Validate required business fields

```
SELECT
    CUSTOM_ACTIVITY,
    COUNT(*) AS TOTAL_ROWS,
    COUNT_IF(CLOSER_EMAIL IS NOT NULL) AS CLOSER_ROWS,
    COUNT_IF(SETTER IS NOT NULL) AS SETTER_ROWS,
    COUNT_IF(OFFER_PRESENTED IS NOT NULL) AS OFFER_ROWS,
    COUNT_IF(STRATEGY_CALL_OUTCOME IS NOT NULL) AS OUTCOME_ROWS
FROM GOLD.ALL_STRATEGIES_DETAILS
GROUP BY CUSTOM_ACTIVITY
ORDER BY CUSTOM_ACTIVITY;

```
# Confirm no row multiplication
```
SELECT
    (SELECT COUNT(*)
     FROM SILVER.LEADS_ACTIVITIES_SUMMARY
     WHERE CUSTOM_ACTIVITY IN (
         '5) Strategy Call',
         '6) Strategy Call Follow Up'
     )) AS SILVER_ROWS,

    (SELECT COUNT(*)
     FROM GOLD.ALL_STRATEGIES_DETAILS) AS GOLD_ROWS;

```
