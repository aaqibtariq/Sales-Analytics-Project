# create table

```
CREATE OR REPLACE TABLE SILVER.LEADS_ACTIVITIES_SUMMARY_TRANSIENT AS
SELECT
    ACTIVITY_ID,
    LEAD_ID,
    CONTACT_ID,
    USER_ID,
    TYPE,
    TEXT,
    STATUS_CHANGE,
    CREATED_BY_NAME,
    UPDATED_BY_NAME,
    ATTENDEES,
    CUSTOM_ACTIVITY_ID,
    MEETING_STARTS_AT,
    MEETING_END_AT,
    MEETING_STATUS,
    ACTIVITY_AT,
    DATE_CREATED,
    DATE_UPDATED,
    CREATED_BY_USER,
    UPDATED_BY_USER,
    INSERT_DATE,
    UPDATE_DATE,
    MD5_HASH
FROM SILVER.LEAD_ACTIVITIES_PROCESSED;


```

# validate 

```

SELECT COUNT(*) FROM SILVER.LEADS_ACTIVITIES_SUMMARY_TRANSIENT;

SELECT *
FROM SILVER.LEADS_ACTIVITIES_SUMMARY_TRANSIENT
LIMIT 20;


SELECT
    (SELECT COUNT(*)
     FROM SALES_ANALYTICS_DB.SILVER.LEAD_ACTIVITIES_PROCESSED)
        AS PROCESSED_ROWS,

    (SELECT COUNT(*)
     FROM SALES_ANALYTICS_DB.SILVER.LEADS_ACTIVITIES_SUMMARY_TRANSIENT)
        AS SUMMARY_TRANSIENT_ROWS;


        SELECT COUNT(*) AS MISSING_FROM_TRANSIENT
FROM SALES_ANALYTICS_DB.SILVER.LEAD_ACTIVITIES_PROCESSED p
LEFT JOIN SALES_ANALYTICS_DB.SILVER.LEADS_ACTIVITIES_SUMMARY_TRANSIENT t
    ON p.LEAD_ID = t.LEAD_ID
   AND p.ACTIVITY_ID = t.ACTIVITY_ID
WHERE t.ACTIVITY_ID IS NULL;

SELECT COUNT(*) AS EXTRA_IN_TRANSIENT
FROM SALES_ANALYTICS_DB.SILVER.LEADS_ACTIVITIES_SUMMARY_TRANSIENT t
LEFT JOIN SALES_ANALYTICS_DB.SILVER.LEAD_ACTIVITIES_PROCESSED p
    ON t.LEAD_ID = p.LEAD_ID
   AND t.ACTIVITY_ID = p.ACTIVITY_ID
WHERE p.ACTIVITY_ID IS NULL;

SELECT
    COUNT_IF(LEAD_ID IS NULL) AS NULL_LEAD_IDS,
    COUNT_IF(ACTIVITY_ID IS NULL) AS NULL_ACTIVITY_IDS,
    COUNT_IF(MD5_HASH IS NULL) AS NULL_MD5_HASHES
FROM SALES_ANALYTICS_DB.SILVER.LEADS_ACTIVITIES_SUMMARY_TRANSIENT;

SELECT
    LEAD_ID,
    ACTIVITY_ID,
    COUNT(*) AS DUPLICATE_COUNT
FROM SALES_ANALYTICS_DB.SILVER.LEADS_ACTIVITIES_SUMMARY_TRANSIENT
GROUP BY
    LEAD_ID,
    ACTIVITY_ID
HAVING COUNT(*) > 1;

```
