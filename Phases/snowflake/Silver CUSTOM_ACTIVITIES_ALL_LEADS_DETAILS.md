# create table 

```

CREATE OR REPLACE TABLE SILVER.CUSTOM_ACTIVITIES_ALL_LEADS_DETAILS (
    LEAD_ID STRING,
    ACTIVITY_ID STRING,
    ACTIVITY_AT TIMESTAMP_NTZ,
    WRAPPED_ACTIVITIES VARIANT,
    INSERT_DATE TIMESTAMP_NTZ,
    UPDATE_DATE TIMESTAMP_NTZ,
    MD5_HASH STRING
);

```

# inset

```
INSERT INTO SILVER.CUSTOM_ACTIVITIES_ALL_LEADS_DETAILS
(
    LEAD_ID,
    ACTIVITY_ID,
    ACTIVITY_AT,
    WRAPPED_ACTIVITIES,
    INSERT_DATE,
    UPDATE_DATE,
    MD5_HASH
)
SELECT
    LEAD_ID,
    ACTIVITY_ID,
    ACTIVITY_AT,
    OBJECT_CONSTRUCT_KEEP_NULL(
        'CUSTOM_ACTIVITY_ID', CUSTOM_ACTIVITY_ID,
        'TYPE', TYPE,
        'TEXT', TEXT,
        'STATUS_CHANGE', STATUS_CHANGE,
        'CREATED_BY_NAME', CREATED_BY_NAME,
        'UPDATED_BY_NAME', UPDATED_BY_NAME,
        'ATTENDEES', ATTENDEES,
        'MEETING_STARTS_AT', MEETING_STARTS_AT,
        'MEETING_END_AT', MEETING_END_AT,
        'MEETING_STATUS', MEETING_STATUS,
        'DATE_CREATED', DATE_CREATED,
        'DATE_UPDATED', DATE_UPDATED,
        'CREATED_BY_USER', CREATED_BY_USER,
        'UPDATED_BY_USER', UPDATED_BY_USER
    ) AS WRAPPED_ACTIVITIES,
    CURRENT_TIMESTAMP(),
    CURRENT_TIMESTAMP(),
    MD5(
        COALESCE(LEAD_ID, '') || '|' ||
        COALESCE(ACTIVITY_ID, '') || '|' ||
        COALESCE(CUSTOM_ACTIVITY_ID, '') || '|' ||
        COALESCE(ACTIVITY_AT::STRING, '')
    ) AS MD5_HASH
FROM SILVER.LEAD_ACTIVITIES_PROCESSED
WHERE CUSTOM_ACTIVITY_ID IS NOT NULL;


```

# validate 

```

SELECT COUNT(*)
FROM SILVER.CUSTOM_ACTIVITIES_ALL_LEADS_DETAILS;

SELECT *
FROM SILVER.CUSTOM_ACTIVITIES_ALL_LEADS_DETAILS
LIMIT 20;

SELECT
    (
        SELECT COUNT(*)
        FROM SALES_ANALYTICS_DB.SILVER.LEAD_ACTIVITIES_PROCESSED
        WHERE CUSTOM_ACTIVITY_ID IS NOT NULL
    ) AS EXPECTED_ROWS,

    (
        SELECT COUNT(*)
        FROM SALES_ANALYTICS_DB.SILVER.CUSTOM_ACTIVITIES_ALL_LEADS_DETAILS
    ) AS ACTUAL_ROWS;

    SELECT
    LEAD_ID,
    ACTIVITY_ID,
    COUNT(*) AS DUPLICATE_COUNT
FROM SALES_ANALYTICS_DB.SILVER.CUSTOM_ACTIVITIES_ALL_LEADS_DETAILS
GROUP BY
    LEAD_ID,
    ACTIVITY_ID
HAVING COUNT(*) > 1;

SELECT
    COUNT_IF(LEAD_ID IS NULL) AS NULL_LEAD_IDS,
    COUNT_IF(ACTIVITY_ID IS NULL) AS NULL_ACTIVITY_IDS,
    COUNT_IF(WRAPPED_ACTIVITIES IS NULL) AS NULL_WRAPPED_ACTIVITIES,
    COUNT_IF(MD5_HASH IS NULL) AS NULL_MD5_HASHES
FROM SALES_ANALYTICS_DB.SILVER.CUSTOM_ACTIVITIES_ALL_LEADS_DETAILS;

SELECT
    LEAD_ID,
    ACTIVITY_ID,
    ACTIVITY_AT,
    WRAPPED_ACTIVITIES,
    MD5_HASH
FROM SALES_ANALYTICS_DB.SILVER.CUSTOM_ACTIVITIES_ALL_LEADS_DETAILS
LIMIT 20;
```
